cmake_minimum_required(VERSION 3.24)
project(SerialBroker VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "SerialBroker currently requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /permissive- /utf-8 /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(serial_core STATIC src/win_io.cpp src/session.cpp src/broker.cpp)
target_include_directories(serial_core PUBLIC include src)
target_link_libraries(serial_core PUBLIC ws2_32 advapi32)
add_executable(SerialBroker WIN32 src/main.cpp)
target_link_libraries(SerialBroker PRIVATE serial_core comctl32 comdlg32)
add_executable(serialctl src/cli.cpp)
target_link_libraries(serialctl PRIVATE serial_core)
install(TARGETS SerialBroker serialctl RUNTIME DESTINATION .)
